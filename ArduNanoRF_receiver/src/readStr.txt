// #include <SPI.h>
// #include <nRF24L01.h>
#include <RF24.h>
#include "Arduino.h"
#include <LiquidCrystal_I2C.h>  // Include LiquidCrystal_I2C library 

RF24 radio(7, 8); // CE, CSN

const byte address[6] = "00001";
unsigned long myTime;

LiquidCrystal_I2C lcd(0x27, 16, 4);  // Configure LiquidCrystal_I2C library with 0x27 address, 16 columns and 2 rows
//sda #a4, scl #a5

void setup() {
  // Serial.begin(9600);
  radio.begin();
  radio.openReadingPipe(0, address);
  radio.setPALevel(RF24_PA_MIN);
  radio.startListening();

  lcd.init();                            // Initialize I2C LCD module
  lcd.backlight();  
}

int i = 0;
void loop() 
{
  if (radio.available())
  {
    char text[50] = "";
    // char temp_buf[50];
    
    //myTime = millis();
    radio.read(&text, sizeof(text));
    //Serial.print("Hello world "); 
    // Serial.print("Read, ");   
    // Serial.println(text); 
    //delay(300);   
    if (text !="\0")
    {
      lcd.setCursor(0,0);
      lcd.print(F("Read            ")); 
      lcd.setCursor(0,1);
      lcd.print(text); /*prints humidity value*/
      int len=strlen(text);
      for (int j = 0; j < len; j++) 
      { 
        lcd.setCursor(j,2);
        lcd.print(text[j]);
      }
      int num5= (text[3] -'0') * 10000;  //ASCii code '0' is 48, ex; int 2 is ASCii 50, so 50-48 = 2
      int num4= (text[4] -'0') * 1000;
      int num3= (text[5] -'0') * 100;
      int num2= (text[6] -'0') * 10;
      int num1= text[7] -'0'; 
      int total = num1 + num2 +num3+ num4+ num5;
      lcd.setCursor(0,3);
      lcd.print(total);
    }
    else
    {
      lcd.clear();
      lcd.setCursor(0,0);
      lcd.print(F("No Data from TX")); /*prints humidity value*/
    }
  }
  else
  {
    lcd.clear();
    lcd.setCursor(0,0);
    lcd.print(F("Radio Not available                                                                                                     ")); /*prints humidity value*/
  }
  delay(500);
}
