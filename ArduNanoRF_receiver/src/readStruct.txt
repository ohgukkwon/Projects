// #include <SPI.h>
// #include <nRF24L01.h>
#include <RF24.h>
#include "Arduino.h"
#include <LiquidCrystal_I2C.h>  // Include LiquidCrystal_I2C library 
#include <stdio.h>

RF24 radio(7, 8); // CE, CSN

//const byte address1[6] = "00001";
const byte address2[6] = "00002";
//unsigned long myTime;

LiquidCrystal_I2C lcd(0x27, 20, 4);  // Configure LiquidCrystal_I2C library with 0x27 address, 16 columns and 2 rows
//sda #a4, scl #a5
struct readData {
  int tx_id;
  float t_6672;
  int cnt;
  int h ;//= dht.readHumidity(); /*float variable that stores humidity value*/
  float t; // = dht.readTemperature(); /*float variable that store temperature in Celsius*/
  float f; // = dht.readTemperature(true); /*variable to store temperature in Fahrenheit*/
} readData;

struct readData1 {
  int tx_id;
  float t_6672;
  int cnt;
  int h ;//= dht.readHumidity(); /*float variable that stores humidity value*/
  float t; // = dht.readTemperature(); /*float variable that store temperature in Celsius*/
  float f; // = dht.readTemperature(true); /*variable to store temperature in Fahrenheit*/
} readData1;

struct readData2 {
  int tx_id;
  float t_6672;
  int cnt;
  int h ;//= dht.readHumidity(); /*float variable that stores humidity value*/
  float t; // = dht.readTemperature(); /*float variable that store temperature in Celsius*/
  float f; // = dht.readTemperature(true); /*variable to store temperature in Fahrenheit*/
} readData2;

void radio_init();
void rf_read();

void setup() {
  // Serial.begin(9600);
  radio.begin();
  //radio.openReadingPipe(0, address1);  //1st RF
  radio.openReadingPipe(1, address2);  //2nd RF
  radio.setPALevel(RF24_PA_MIN);
  radio.startListening();
  radio.printDetails(); 
  // radio.setAutoAck();

  lcd.init();                            // Initialize I2C LCD module
  lcd.backlight();  
}

void loop() 
{
  if (radio.available())
  {
    // float text[50];
    radio.read(&readData, sizeof(readData));  //read data save to struct
    rf_read();
    // radio.read(&readData, sizeof(readData));  //read data save to struct
    // rf_read();

   //from tx01 
    lcd.setCursor(0,0);
    lcd.print(readData1.tx_id);
    lcd.print(F(":"));
    lcd.print(readData1.t_6672, 1);     //float decimal point change ex - 26.00 ->26.0
    lcd.print(F("'C, CNT:"));
    lcd.print(readData1.cnt);

    // from tx02
    lcd.setCursor(0,1);
    lcd.print(readData2.tx_id);
    lcd.print(F(":"));
    lcd.print(readData2.t_6672, 1);     //float decimal point change ex - 26.00 ->26.0
    lcd.print(F("'C, CNT:"));
    lcd.print(readData2.cnt);

  }
  else
  {
    lcd.clear();
    lcd.setCursor(0,0);
    lcd.print("Network Fault");
    
    for (int i=3; i>=0; i--)
    {
      lcd.setCursor(0,0);
      lcd.print("Network reset");
      lcd.setCursor(0,1);
      lcd.print(i);
      delay(1000);
    }
    radio.powerDown();
    radio_init();
    radio.powerUp();
    lcd.clear();
  }
  delay(500);
}

void radio_init()
{
  radio.begin();
  radio.openReadingPipe(1, address2);
  radio.setPALevel(RF24_PA_MIN);
  radio.startListening();
  radio.printDetails(); 
  delay(1000);

}

void rf_read()
{
  if (readData.tx_id == 0)
  {
    readData1.tx_id   = readData.tx_id ;
    readData1.t_6672  = readData.t_6672;
    readData1.cnt     = readData.cnt   ;
    readData1.h       = readData.h     ;//= dht.readHumidity(); /*float variable that stores humidity value*/
    readData1.t       = readData.t     ; // = dht.readTemperature(); /*float variable that store temperature in Celsius*/
    readData1.f       = readData.f; // = dht.readTemperature(true); /*variable to store temperature in Fahrenheit*/
  }
  if (readData.tx_id == 2)
  {
    readData2.tx_id   = readData.tx_id ;
    readData2.t_6672  = readData.t_6672;
    readData2.cnt     = readData.cnt   ;
    readData2.h       = readData.h     ;//= dht.readHumidity(); /*float variable that stores humidity value*/
    readData2.t       = readData.t     ; // = dht.readTemperature(); /*float variable that store temperature in Celsius*/
    readData2.f       = readData.f     ; // = dht.readTemperature(true); /*variable to store temperature in Fahrenheit*/
  }

}
