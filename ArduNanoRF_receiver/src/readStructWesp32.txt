#include <SPI.h>
#include <nRF24L01.h>
#include <RF24.h>
#include <LiquidCrystal_I2C.h>  // Include LiquidCrystal_I2C library 
LiquidCrystal_I2C lcd(0x27, 20, 4);  // Configure LiquidCrystal_I2C library with 0x27 address, 16 columns and 2 rows
//sda #a4, scl #a5

RF24 radio(7, 8); // CE-7, CSN-8, SCK-13, MISO-12, MOSI-11
//NRF24L01 power is 3.3V --

const byte address[6] = "00001";
//const byte address2[6] = "00002";

struct ReadData{
  int tx_id;
  char sensor_data[3];
  int cnt;
  float max6675_c;
  int dht_hum;
  float dht_c;
  float dht_f; 
};
ReadData readData;

void radio_init();

void setup() {
  Serial.begin(9600);
  radio.begin();
  radio.openReadingPipe(1, address);
  //radio.openReadingPipe(2, address2);
  radio.setPALevel(RF24_PA_MIN);
  radio.startListening();
  
  lcd.clear();
  lcd.init();                            // Initialize I2C LCD module
  lcd.backlight();  
}

void loop() {
  if (radio.available()) 
  {
    //char text[32] = "";
    radio.read(&readData, sizeof(readData));
    // Serial.print("Rx: ");
    // Serial.println(text);
    if (readData.tx_id ==1)
    {
      lcd.setCursor(0,0);
      lcd.print("#");
      lcd.print(readData.tx_id);
      lcd.print(",Ard,");
      
      lcd.print(readData.cnt);
      lcd.print("                   ");
    }

    radio.read(&readData, sizeof(readData));
    if (readData.tx_id == 2)
    {
      lcd.setCursor(0,1);
      lcd.print("#");
      lcd.print(readData.tx_id);
      lcd.print(",ESP,");

      lcd.print(readData.cnt/255);         //esp32 integer convert
      lcd.print("                   ");
      //Serial.println(readData.cnt);
    }
    delay(50);
  }
  else
  {
    lcd.clear();
    lcd.setCursor(0,0);
    lcd.print("Network Fault");
    
    for (int i=3; i>=0; i--)
    {
      lcd.setCursor(0,0);
      lcd.print("Network reset");
      lcd.setCursor(0,1);
      lcd.print(i);
      delay(1000);
    }
    radio.powerDown();
    radio_init();
    radio.powerUp();
    lcd.clear();
  }
  delay(500);
}
void radio_init()
{
  radio.begin();
  radio.openReadingPipe(1, address);
  radio.setPALevel(RF24_PA_MIN);
  radio.startListening();
  radio.printDetails(); 
  delay(1000);
}
