void setup() {
  radio.begin();
  radio.openWritingPipe(address);
  radio.openReadingPipe(0, address);
  radio.setPALevel(RF24_PA_MIN);
  radio.stopListening();
  radio.printDetails(); 

  // Serial.begin(9600);
  // Serial.setTimeout(1000);

  lcd.clear();

}
